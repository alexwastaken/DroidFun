postinstall scriptlet:
#!/bin/sh
set -eu
# Package managers already authorize this step. No account/group changes.
if ! /usr/bin/sh /usr/lib/droidfun/setup-robot-access; then
    echo 'DroidFun USB rule installed. Reconnect the robot after installation, or choose Set up robot access in DroidFun.' >&2
fi
exit 0

postuninstall scriptlet:
#!/bin/sh
# Files are owned/removed by the package manager. Never touch user rules.
if [ -S /run/udev/control ]; then
    /usr/bin/timeout 10s /usr/bin/udevadm control --reload-rules || true
fi
exit 0
